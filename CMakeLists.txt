cmake_minimum_required(VERSION 3.21)
project(fltk-dock VERSION 0.1.0 LANGUAGES CXX)
if(APPLE)
    enable_language(OBJCXX)
endif()
option(FLTK_DOCK_BUILD_EXAMPLE "Build the docking example" ${PROJECT_IS_TOP_LEVEL})
option(FLTK_DOCK_BUILD_TESTS "Build docking and tab contract tests" ${PROJECT_IS_TOP_LEVEL})
option(FLTK_DOCK_FETCH_FLTK "Fetch FLTK 1.4.5 if no package is installed" ON)
if(NOT TARGET fltk::fltk)
    find_package(FLTK 1.4.5 CONFIG QUIET)
endif()
if(NOT TARGET fltk::fltk)
    if(NOT FLTK_DOCK_FETCH_FLTK)
        message(FATAL_ERROR "FLTK 1.4.5 or newer is required. Set FLTK_DIR or enable FLTK_DOCK_FETCH_FLTK.")
    endif()
    include(FetchContent)
    foreach(feature TEST EXAMPLES FLUID FLTK_OPTIONS GL FORMS HTML_DOCS PDF_DOCS SHARED_LIBS)
        set(FLTK_BUILD_${feature} OFF CACHE BOOL "" FORCE)
    endforeach()
    FetchContent_Declare(fltk GIT_REPOSITORY https://github.com/fltk/fltk.git GIT_TAG release-1.4.5 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(fltk)
endif()
add_library(fltk-dock STATIC src/fltk_dock.cpp src/fltk_dock_tabs.cpp)
add_library(fltk-dock::fltk-dock ALIAS fltk-dock)
target_compile_features(fltk-dock PUBLIC cxx_std_17)
target_include_directories(fltk-dock PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(fltk-dock PUBLIC fltk::fltk)
if(MSVC)
    target_compile_options(fltk-dock PRIVATE /utf-8 /W4)
endif()
if(FLTK_DOCK_BUILD_EXAMPLE)
    add_executable(fltk-dock-demo examples/demo.cpp)
    target_link_libraries(fltk-dock-demo PRIVATE fltk-dock)
endif()
if(FLTK_DOCK_BUILD_TESTS)
    enable_testing()
    add_executable(fltk-dock-tests tests/contracts.cpp)
    target_link_libraries(fltk-dock-tests PRIVATE fltk-dock fltk::images)
    add_test(NAME dock-contracts COMMAND fltk-dock-tests "${CMAKE_CURRENT_BINARY_DIR}/test-output")
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS fltk-dock EXPORT fltk-dock-targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT fltk-dock-targets NAMESPACE fltk-dock:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fltk-dock)
configure_package_config_file(cmake/fltk-dock-config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/fltk-dock-config.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fltk-dock)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/fltk-dock-config-version.cmake" COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/fltk-dock-config.cmake" "${CMAKE_CURRENT_BINARY_DIR}/fltk-dock-config-version.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fltk-dock)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fltk-dock)
